local function getParts(p)
    local c = p.Character; if not c then return nil, nil, nil end
    local hum = c:FindFirstChildWhichIsA("Humanoid", true)
    local root = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso")
    local head = c:FindFirstChild("Head"); if not head then head = root end
    return root, head, hum
end

local function getAimPart(char)
    if not char then return nil end
    for _, name in ipairs(PART_FALLBACK[CFG.AimbotPart] or {"Head"}) do
        local part = char:FindFirstChild(name)
        if part and part:IsA("BasePart") then return part end
    end
    return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildWhichIsA("BasePart")
end

local function w2s(pos)
    local v, on = Camera:WorldToViewportPoint(pos)
    if v.Z <= 0 then return Vector2.new(v.X, v.Y), false end
    return Vector2.new(v.X, v.Y), true
end

local function screenCenter() return Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2) end

local function isEnemy(p)
    if not CFG.AimbotTeamCheck then return true end
    return p.Team ~= LocalPlayer.Team
end

local _rp = RaycastParams.new(); _rp.FilterType = Enum.RaycastFilterType.Exclude
local function hasLOS(from, to, extra)
    local char = LocalPlayer.Character; local ig = char and {char} or {}
    if extra then table.insert(ig, extra) end
    _rp.FilterDescendantsInstances = ig
    return workspace:Raycast(from, to-from, _rp) == nil
end

local function getAimPos(part)
    if not part then return nil end
    return (part.CFrame * CFrame.new(CFG.HeadOffsetX, CFG.HeadOffsetY, 0)).Position
end

local deadFlags = {}
local humRegistry = {}

local function isAliveRaw(p)
    local char = p.Character; if not char or not char.Parent then return false end
    local hum = char:FindFirstChildWhichIsA("Humanoid", true)
    if not hum or not hum.Parent then return false end
    if hum.Health <= 0 then return false end
    local ok, state = pcall(function() return hum:GetState() end)
    if ok and state == Enum.HumanoidStateType.Dead then return false end
    if char:FindFirstChild("Dead") or char:FindFirstChild("IsDead") then return false end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root and root.Anchored then return false end
    return true
end

local pool = {}
local lootEspPool = {}
local fovCircle = D_Circle(); fovCircle.Color = CFG.ColorFOV

local function alloc(p)
    if pool[p] then return end
    local t = {
        box = D_Square(false, 1), name = D_Text(13), dist = D_Text(11),
        hpBg = D_Square(true, 1), hpBar = D_Square(true, 1), tracer = D_Line(),
    }
    t.box.Color = Color3.new(1,1,1); t.name.Color = CFG.ColorText
    t.dist.Color = Color3.fromRGB(180,180,180); t.hpBg.Color = Color3.new(0,0,0)
    t.tracer.Color = CFG.ColorTracer; pool[p] = t
end

local function free(p)
    if not pool[p] then return end
    for _, d in pairs(pool[p]) do pcall(function() d:Remove() end) end
    pool[p] = nil; deadFlags[p] = nil; humRegistry[p] = nil
end

local function hideAll(e) if not e then return end; for _, d in pairs(e) do d.Visible = false end end

local function renderPlayer(p, e)
    if not isAliveRaw(p) then hideAll(e) return end
    local root, head, hum = getParts(p)
    if not root then local char = p.Character; if char then root = char:FindFirstChildWhichIsA("BasePart") end end
    if not head then head = root end
    if not root or not hum then hideAll(e) return end
    local dist3D = (root.Position - Camera.CFrame.Position).Magnitude
    if dist3D > CFG.ESPMaxDist then hideAll(e) return end
    if CFG.ESPWallCheck and not hasLOS(Camera.CFrame.Position, head.Position, p.Character) then hideAll(e) return end
    
    local topSP, _ = w2s(head.Position + Vector3.new(0, 0.7, 0))
    local botSP, _ = w2s(root.Position - Vector3.new(0, 3.0, 0))
    
    local screenZ = Camera:WorldToViewportPoint(root.Position).Z
    if screenZ <= 0 then hideAll(e) return end

    local bH = math.clamp(math.abs(topSP.Y - botSP.Y), 2, 2000)
    local bW = math.clamp(bH * 0.6, 2, 1000)
    local bX = topSP.X - bW / 2
    local col = isEnemy(p) and CFG.ColorEnemy or CFG.ColorTeammate
    
    e.box.Visible = CFG.ESPBox; e.box.Color = col
    e.box.Position = Vector2.new(bX, topSP.Y); e.box.Size = Vector2.new(bW, bH)
    e.name.Visible = CFG.ESPName; e.name.Color = CFG.ColorText
    e.name.Text = p.Name; e.name.Position = Vector2.new(topSP.X, topSP.Y - 16)
    e.dist.Visible = CFG.ESPDistance
    e.dist.Text = string.format("[%dm]", math.floor(dist3D/4))
    e.dist.Position = Vector2.new(topSP.X, topSP.Y - 28)
    
    local hpR = hum.MaxHealth > 0 and math.clamp(hum.Health/hum.MaxHealth, 0, 1) or 0
    e.hpBg.Visible = CFG.ESPHealth; e.hpBg.Position = Vector2.new(bX - 6, topSP.Y); e.hpBg.Size = Vector2.new(3, bH)
    e.hpBar.Visible = CFG.ESPHealth; e.hpBar.Color = hpR > 0.4 and CFG.ColorHP or CFG.ColorHPLow
    e.hpBar.Position = Vector2.new(bX - 6, topSP.Y + bH - bH * hpR); e.hpBar.Size = Vector2.new(3, math.max(bH * hpR, 1))
    
    e.tracer.Visible = CFG.ESPTracers; e.tracer.Color = CFG.ColorTracer
    e.tracer.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y); e.tracer.To = botSP
end

local function hookCharacter(p, char)
    task.spawn(function()
        local hum; pcall(function() hum = char:WaitForChild("Humanoid", 8) end)
        if not hum then hum = char:FindFirstChildWhichIsA("Humanoid", true) end
        if not hum then return end
        humRegistry[p] = hum
        hum.Died:Connect(function() deadFlags[p] = true end)
        hum.HealthChanged:Connect(function(hp) if hp <= 0 then deadFlags[p] = true end end)
        hum.AncestryChanged:Connect(function() if not hum.Parent then deadFlags[p] = true end end)
        char.AncestryChanged:Connect(function() if not char.Parent then deadFlags[p] = true end end)
    end)
end

local function hookPlayer(p)
    if p.Character then alloc(p); hookCharacter(p, p.Character) end
    p.CharacterAdded:Connect(function(char)
        deadFlags[p] = nil; humRegistry[p] = nil
        char:WaitForChild("HumanoidRootPart", 5)
        alloc(p); hookCharacter(p, char)
    end)
end

for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer then hookPlayer(p) end end
Players.PlayerAdded:Connect(hookPlayer)
Players.PlayerRemoving:Connect(free)

-- AIMBOT LOGIC

local aimbotTarget = nil
local aimbotTargetPlayer = nil
local aimbotToggleActive = false

local function isAimbotFiring()
    if not CFG.AimbotEnabled then return false end
    if CFG.AimbotMode == "Toggle" then return aimbotToggleActive end
    return UserInputService:IsKeyDown(CFG.AimbotKey)
end

local function getClosestTarget()
    local bestPlayer = nil
    local bestPart = nil
    local bestDistance = CFG.AimbotFOV
    local center = screenCenter()

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and isEnemy(p) and isAliveRaw(p) then
            local char = p.Character
            local aimPart = char and getAimPart(char)

            if aimPart and aimPart:IsA("BasePart") then
                local aimPos = getAimPos(aimPart)
                local screenPos, _ = Camera:WorldToViewportPoint(aimPos)

                if screenPos.Z > 0 then
                    local pos2D = Vector2.new(screenPos.X, screenPos.Y)
                    local distance = (pos2D - center).Magnitude

                    if distance <= bestDistance then
                        local wallOK = not CFG.AimbotWallCheck or hasLOS(Camera.CFrame.Position, aimPos, char)
                        if wallOK then
                            bestDistance = distance
                            bestPlayer = p
                            bestPart = aimPart
                        end
                    end
                end
            end
        end
    end
    return bestPlayer, bestPart
end

RunService.Heartbeat:Connect(function()
    if not isAimbotFiring() then
        aimbotTarget = nil; aimbotTargetPlayer = nil
        return
    end

    local player, part = getClosestTarget()
    if player and part then
        aimbotTargetPlayer = player
        aimbotTarget = part
    else
        aimbotTarget = nil; aimbotTargetPlayer = nil
    end
end)

RunService.RenderStepped:Connect(function()
    if not isAimbotFiring() then return end
    local player = aimbotTargetPlayer

    if not player or not isAliveRaw(player) then
        aimbotTarget = nil; aimbotTargetPlayer = nil
        return
    end

    local char = player.Character
    if not char then
        aimbotTarget = nil; aimbotTargetPlayer = nil
        return
    end

    local aimPart = getAimPart(char)
    if not aimPart or not aimPart:IsA("BasePart") then
        aimbotTarget = nil; aimbotTargetPlayer = nil
        return
    end

    local aimPos = getAimPos(aimPart)
    aimbotTarget = aimPart

    local screenPos, _ = Camera:WorldToViewportPoint(aimPos)
    if screenPos.Z <= 0 then return end

    local center = screenCenter()
    local dx = screenPos.X - center.X
    local dy = screenPos.Y - center.Y
    local distance = math.sqrt(dx * dx + dy * dy)

    if distance > CFG.AimbotFOV then return end
    if CFG.AimbotWallCheck and not hasLOS(Camera.CFrame.Position, aimPos, char) then return end

    local smooth = math.clamp(CFG.AimbotSmooth, 0.01, 1)
    dx = dx * smooth
    dy = dy * smooth

    if math.abs(dx) < 0.05 then dx = 0 end
    if math.abs(dy) < 0.05 then dy = 0 end

    if (dx ~= 0 or dy ~= 0) and mousemoverel then
        mousemoverel(dx, dy)
    end
end)

local reAllocTimer = 0

RunService.RenderStepped:Connect(function(dt)
    local center = screenCenter()

    fovCircle.Visible = CFG.ShowFOVCircle
    fovCircle.Position = center
    fovCircle.Radius = CFG.AimbotFOV
    fovCircle.Color = CFG.ColorFOV

    reAllocTimer = reAllocTimer + (dt or 0)
    if reAllocTimer > 0.5 then
        reAllocTimer = 0
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                if not pool[p] then alloc(p) end
                if p.Character and not humRegistry[p] then hookCharacter(p, p.Character) end
            end
        end
    end

    for p, _ in pairs(pool) do
        if not p.Parent then free(p) end
    end

    -- 1. Loot ESP (มองเห็นของบนพื้น WarzLoot)
    local lootFolder = workspace:FindFirstChild("WarzLoot")
    local activeLootItems = {}
    if CFG.ESPLoot and lootFolder then
        for _, item in ipairs(lootFolder:GetChildren()) do
            if item:IsA("BasePart") or item:IsA("Model") then
                local part = item:IsA("Model") and (item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")) or item
                if part then
                    activeLootItems[item] = part
                end
            end
        end
    end

    for itemObj, ui in pairs(lootEspPool) do
        if not activeLootItems[itemObj] then
            ui.Visible = false
            pcall(function() ui:Remove() end)
            lootEspPool[itemObj] = nil
        end
    end

    for itemObj, part in pairs(activeLootItems) do
        local ui = lootEspPool[itemObj]
        if not ui then
            ui = Drawing.new("Text")
            ui.Size = 11
            ui.Center = true
            ui.Outline = true
            ui.OutlineColor = Color3.new(0,0,0)
            ui.Color = CFG.ColorLoot
            lootEspPool[itemObj] = ui
        end

        local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
        local rootChar = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local dist = rootChar and (part.Position - rootChar.Position).Magnitude or 0

        if onScreen and screenPos.Z > 0 and dist <= CFG.ESPMaxDist then
            ui.Visible = true
            ui.Position = Vector2.new(screenPos.X, screenPos.Y)
            ui.Text = string.format("📦 [%s] (%dm)", itemObj.Name, math.floor(dist/3.5))
        else
            ui.Visible = false
        end
    end

    if not CFG.ESPEnabled then
        for _, e in pairs(pool) do hideAll(e) end
        return
    end

    for p, e in pairs(pool) do
        renderPlayer(p, e)
    end
end)

-- แก้ไข Slider ไม่ให้ค้างและไม่ให้แถบสีเขียวยืดทะลุจอ
local function makeSlider(labelText, min, max, current, step, onChange, parent)
    step = step or 1
    local f = Instance.new("Frame"); f.Size = UDim2.new(1, 0, 0, 56); f.BackgroundColor3 = T.CARD2; f.Parent = parent
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)
    local lbl = Instance.new("TextLabel"); lbl.Size = UDim2.new(0.6, 0, 0, 24); lbl.Position = UDim2.new(0, 12, 0, 6)
    lbl.BackgroundTransparency = 1; lbl.Text = labelText; lbl.TextColor3 = T.TEXT; lbl.Font = Enum.Font.Gotham; lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left; lbl.Parent = f
    local valLbl = Instance.new("TextLabel"); valLbl.Size = UDim2.new(0.4, -12, 0, 24); valLbl.Position = UDim2.new(0.6, 0, 0, 6)
    valLbl.BackgroundTransparency = 1; valLbl.Text = string.format(step<1 and "%.2f" or "%d", current)
    valLbl.TextColor3 = T.ACCENTLIT; valLbl.Font = Enum.Font.GothamBold; valLbl.TextSize = 11
    valLbl.TextXAlignment = Enum.TextXAlignment.Right; valLbl.Parent = f
    local track = Instance.new("Frame"); track.Size = UDim2.new(1, -24, 0, 6); track.Position = UDim2.new(0, 12, 0, 38)
    track.BackgroundColor3 = T.TRACK; track.Parent = f
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)
    
    local initialFrac = math.clamp((current - min) / (max - min), 0, 1)
    local fill = Instance.new("Frame"); fill.Size = UDim2.new(initialFrac, 0, 1, 0); fill.BackgroundColor3 = T.ACCENT; fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)
    
    local dragging = false
    local function upd(px)
        local absPos = track.AbsolutePosition.X
        local absSize = track.AbsoluteSize.X
        if absSize <= 0 then return end
        local rel = math.clamp((px - absPos) / absSize, 0, 1)
        local v = math.floor((min + rel * (max - min)) / step + 0.5) * step
        v = math.clamp(v, min, max)
        local frac = math.clamp((v - min) / (max - min), 0, 1)
        fill.Size = UDim2.new(frac, 0, 1, 0)
        valLbl.Text = string.format(step<1 and "%.2f" or "%d", v)
        onChange(v)
    end
    
    local hb = Instance.new("TextButton"); hb.Size = UDim2.new(1, 20, 1, 20); hb.Position = UDim2.new(0, -10, 0, -10); hb.BackgroundTransparency = 1; hb.Text = ""; hb.Parent = track
    
    hb.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            upd(input.Position.X)
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            upd(input.Position.X)
        end
    end)
    
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
